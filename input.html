Database design.

RLS enabled · all tables 
-- Order core · money in cents, timestamptz everywhere
CREATE TABLE orders (
  id           uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  merchant_id  uuid REFERENCES merchants(id),
  customer_id  uuid REFERENCES users(id),
  status       order_status DEFAULT 'PLACED',
  subtotal_c   integer NOT NULL,
  delivery_fee_c integer DEFAULT 0,
  eta_ms      integer,
  idempotency_key text UNIQUE,
  placed_at   timestamptz DEFAULT now()
);

CREATE INDEX orders_eta_idx ON orders (eta_ms) WHERE status = 'OUT_FOR_DELIVERY';

-- Prevent double-serve at the cabinet
CREATE POLICY serve_safe ON orders FOR UPDATE USING (auth.uid() = customer_id);
